##sum of natural number
#include <iostream>
using namespace std;

int sum(int n) {
    int sum = 0;
    sum += (n * (n + 1)) / 2;
    return sum;
}

int main() {
    int num;
    cout << "Enter the number to find is sum of n natural number: ";
    cin >> num;
    cout << sum(num) << endl;
    return 0;
}
